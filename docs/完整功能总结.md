```bash
# 运行工具测试

```

### 2. 追踪系统

```bash
# 查看最近 10 条追踪记录
.venv/bin/python scripts/view_user_actions.py

# 查看统计信息
.venv/bin/python scripts/view_user_actions.py --stats

# 查看特定 action_id
.venv/bin/python scripts/view_user_actions.py --action-id abc123

# 运行追踪系统测试
.venv/bin/python -m pytest tests/unit/test_user_action_tracker.py -v
```

### 3. 集成到现有系统

```python
from src.agents.tools import WebSearchTool, ImageRetrievalTool, LaTeXRendererTool
from src.agents.tools.base import ToolExecutor

# 初始化工具
tools = [
    WebSearchTool(),
    ImageRetrievalTool(),
    LaTeXRendererTool(),
]
tool_executor = ToolExecutor(tools)

# 创建 orchestrator（追踪系统已自动集成）
orchestrator = LessonOrchestrator(
    ...,
    tool_executor=tool_executor,
)

# 生成教案（自动追踪）
result = orchestrator.run(
    topic="量子计算",
    template_category="comprehensive",
    conversation_state=state,
)

# 查看工具结果
print(f"工具执行: {len(result['tool_results'])} 个")

# 查看追踪记录
from src.observability.user_action_tracker import get_tracker
tracker = get_tracker()
stats = tracker.get_statistics()
print(f"总请求数: {stats['total_actions']}")
```
